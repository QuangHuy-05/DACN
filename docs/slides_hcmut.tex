\begin{frame}{Problem Statement}
  \vspace{0.05cm}
  \centerline{\large\textbf{Mô hình hóa 4 Tình huống Không gian - Thời gian (A, B, C, D)}}
  \vspace{0.25cm}

  \begin{itemize}
    \setlength{\itemsep}{7pt}
    \item \textbf{Tình huống A (Ánh xạ xuôi Cũ $\to$ Mới):} Đơn vị hành chính cũ chuyển sang mới theo quan hệ: $1 - 1$ (giữ nguyên), $N - 1$ (hợp nhất), $1 - N$ (chia tách), $M - N$ (chia tách và tái cơ cấu phức tạp).
    \item \textbf{Tình huống B (Ánh xạ ngược Mới $\to$ Cũ):} Truy xuất lịch sử địa chỉ từ hệ mới phục vụ lưu trữ hồ sơ pháp lý (Bài toán phân rã $1 \to N$ bất định nếu thiếu số nhà).
    \item \textbf{Tình huống C (Địa chỉ Lai):} Chuỗi địa chỉ pha trộn: C1 (Phường mới + Huyện cũ + Tỉnh cũ), C2 (Phường mới + Huyện cũ + Tỉnh mới), C3 (Phường cũ + Huyện cũ + Tỉnh mới).
    \item \textbf{Tình huống D (Nhập nhằng Không gian):} Đồng âm tên xã/phường diện rộng, mất neo phân cấp khi bị khuyết quận/huyện hoặc gặp nhiễu OCR.
  \end{itemize}
\end{frame}

\begin{frame}{Work 1}
  \vspace{0.05cm}
  \centerline{\large\textbf{Phương pháp Tiếp cận Dựa trên Quy tắc (Rule-based / Gazetteer)}}
  \vspace{0.2cm}

  \begin{itemize}
    \setlength{\itemsep}{4pt}
    \item \textbf{Đại diện tiêu biểu:} Thư viện \texttt{vietnamadminunits} (bộ từ điển hành chính kết hợp Trie / Regex).
    \item \textbf{Cơ chế hoạt động:} Tra cứu từ khóa hành chính ngược từ phải sang trái (Tỉnh $\to$ Huyện $\to$ Xã), bóc tách phần còn lại thành Số nhà và Tên đường.
    \item \textbf{Ưu điểm:}
      \begin{itemize}
        \setlength{\itemsep}{2pt}
        \item Tốc độ xử lý cực nhanh, bộ nhớ thấp.
        \item Độ chính xác tuyệt đối ($97{,}3\%$) khi chuỗi đầu vào sạch chuẩn, đúng chính tả tiếng Việt.
      \end{itemize}
    \item \textbf{Nhược điểm cốt tử:}
      \begin{itemize}
        \setlength{\itemsep}{2pt}

% SLIDE 21: METRICS (Evaluation Metrics)
% ==========================================
\begin{frame}{Metrics}
  \vspace{0.05cm}
  \centerline{\large\textbf{Hệ thống Chỉ số Đo lường Độc lập \& Khắt khe}}
  \vspace{0.2cm}

  \begin{itemize}
    \setlength{\itemsep}{5pt}
    \item \textbf{1. Độ chính xác Tuyệt đối (Exact Match - EM):}
      $$\text{Exact Match}(\hat{\mathbf{y}}) = \bigwedge_{f \in F^*} \mathbb{I}(\text{Norm}(\hat{y}_f) = \text{Norm}(y_f^*)) \qquad (\text{với } F^* = \{5 \text{ trường cốt lõi}\})$$
    \item \textbf{2. Macro-averaged Token F1 từng trường:}
      $$\text{F1}_f = \frac{2 \cdot \text{Precision}_f \cdot \text{Recall}_f}{\text{Precision}_f + \text{Recall}_f}, \qquad \text{Avg\_F1} = \frac{1}{|F^*|} \sum_{f \in F^*} \text{F1}_f$$
    \item \textbf{3. Tỷ lệ Bỏ sót (Missing Rate) \& Tỷ lệ Ảo giác (Hallucination Rate):}
      \begin{itemize}
        \setlength{\itemsep}{2.5pt}
        \item \textbf{Missing Rate:} Nhãn thực tế $y_f^* \ne \emptyset$ nhưng mô hình dự đoán $\hat{y}_f = \emptyset$.
        \item \textbf{Hallucination Rate:} Nhãn thực tế $y_f^* = \emptyset$ (cấp Quận ở hệ 2 cấp) nhưng mô hình tự bịa $\hat{y}_f \ne \emptyset$.
      \end{itemize}
  \end{itemize}
\end{frame}

    \item \textbf{Khối lượng gán nhãn thủ công lớn (Label Studio):} Tập hóa đơn thật Viet-Receipt ($\sim$2.000 mẫu) và tập mốc định vị ($\sim$500 mẫu) đòi hỏi nhiều nhân lực gán nhãn span 11 lớp và đo độ đồng thuận Cohen's Kappa.\\
      $\to$ \textit{Giải pháp tháo gỡ:} Áp dụng cơ chế tiền gán nhãn bán tự động (Weak supervision / Regex pre-annotation) để tăng tốc độ gán nhãn tay.
    \item \textbf{Mơ hồ trong quan hệ chia tách $M - N$ khi thiếu số nhà:} Điểm gãy lớn nhất là khi một xã cũ phân rã thành nhiều phường mới (như 12 ca sai ở Data 07), nếu chuỗi đầu vào không có số nhà thì không thể xác định ranh giới bằng luật.\\
      $\to$ \textit{Giải pháp tháo gỡ:} Xây dựng mô-đun Graph RAG kết hợp tra cứu tri thức địa phương.
    \item \textbf{Ràng buộc tài nguyên tính toán (GPU 16GB):} Giới hạn bộ nhớ trên GPU đơn (Google Colab / Kaggle) khi huấn luyện các mô hình Pretrained Language Models (PhoBERT/ViBERT).\\
      $\to$ \textit{Giải pháp tháo gỡ:} Sử dụng kỹ thuật LoRA/Adapter và mixed-precision (FP16) để tối ưu hóa VRAM.
  \end{itemize}
  }
